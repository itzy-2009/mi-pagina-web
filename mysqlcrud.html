1. Crear Base de Datos y Tablas

-- Crear y activar la base de datos
CREATE DATABASE escuelakaren;
USE escuelakaren;

-- Crear una tabla con su clave primaria
CREATE TABLE alumnos (
    id_alumno INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL,
    edad INT,
    correo VARCHAR(50)
);

2. Insertar Registros

INSERT INTO alumnos (nombre, edad, correo) 
VALUES ('Karen Itzel', 17, 'karen@email.com');

INSERT INTO alumnos (nombre, edad, correo) 
VALUES ('Carlos Pérez', 18, 'carlos@email.com');

3. Consultas Básicas (SELECT)

-- Consultar todos los registros
SELECT * FROM alumnos;

-- Consultar con una condición de filtrado
SELECT nombre, correo FROM alumnos WHERE edad >= 18;

4. Actualizar Registros (UPDATE)

-- Modificar datos existentes utilizando WHERE por seguridad
UPDATE alumnos 
SET correo = 'nuevo_correo@email.com' 
WHERE id_alumno = 1;

5. Eliminar Registros (DELETE)

-- Borrar filas específicas de una tabla
DELETE FROM alumnos 
WHERE id_alumno = 2;
